{{- /* "Add to calendar" control for an event page. The live site offers a single .ics download. This offers the same file plus two direct links, because most people here are adding to Google or to Office 365 and a downloaded file is an extra step for them. Built on
/ rather than a scripted menu: it opens, closes and takes keyboard focus with no JavaScript, so it still works if a script fails. Times come from event-datetimes.html, the same partial the .ics feed uses, so no two routes into a calendar can disagree. Renders nothing for an undated event -- the six recurring sessions have no date to add. */ -}} {{- $dt := partial "event-datetimes.html" . -}} {{- if $dt.ok -}} {{- $loc := .Params.event_location | default "" -}} {{- $ics := "" -}} {{- with .OutputFormats.Get "calendar" }}{{ $ics = .RelPermalink }}{{ end -}} {{- $gdates := printf "%s/%s" $dt.start $dt.end -}} {{- $gq := slice "action" "TEMPLATE" "text" .Title "dates" $gdates "location" $loc "details" .Permalink -}} {{- if not $dt.allday }}{{ $gq = $gq | append "ctz" $dt.tz }}{{ end -}} {{- $google := printf "https://calendar.google.com/calendar/render?%s" (querify $gq) -}} {{- $outlook := "" -}} {{- if $dt.allday -}} {{- /* Outlook wants YYYY-MM-DD; the partial hands back YYYYMMDD. */ -}} {{- $hyphen := printf "%s-%s-%s" (substr $dt.start 0 4) (substr $dt.start 4 2) (substr $dt.start 6 2) -}} {{- $hyphenEnd := printf "%s-%s-%s" (substr $dt.end 0 4) (substr $dt.end 4 2) (substr $dt.end 6 2) -}} {{- $outlook = printf "https://outlook.office.com/calendar/deeplink/compose?%s" (querify "path" "/calendar/action/compose" "rru" "addevent" "allday" "true" "subject" .Title "startdt" $hyphen "enddt" $hyphenEnd "location" $loc "body" .Permalink) -}} {{- else if $dt.isostart -}} {{- $outlook = printf "https://outlook.office.com/calendar/deeplink/compose?%s" (querify "path" "/calendar/action/compose" "rru" "addevent" "subject" .Title "startdt" $dt.isostart "enddt" $dt.isoend "location" $loc "body" .Permalink) -}} {{- end -}}
